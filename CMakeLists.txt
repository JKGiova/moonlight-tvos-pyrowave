cmake_minimum_required(VERSION 3.20)
project(moonlight_pyrowave_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(nonary-framing STATIC native/protocol/pyrowaveframing.cpp)
target_include_directories(nonary-framing PUBLIC native/protocol)
target_compile_options(nonary-framing PRIVATE "$<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra;-Wpedantic>")
add_executable(framing-check tools/framing_check.cpp)
target_link_libraries(framing-check PRIVATE nonary-framing)
include(CTest)
if(BUILD_TESTING)
  add_executable(client-runtime-tests tests/client_runtime_tests.cpp)
  target_include_directories(client-runtime-tests PRIVATE native/client)
  add_test(NAME client-runtime COMMAND client-runtime-tests)
  add_executable(framing-tests tests/framing_tests.cpp)
  target_link_libraries(framing-tests PRIVATE nonary-framing)
  add_test(NAME framing COMMAND framing-tests)
  add_executable(codec-layout-tests tests/codec_layout_tests.cpp third_party/pyrowave/metal/pyrowave_bitstream.cpp)
  target_include_directories(codec-layout-tests PRIVATE third_party/pyrowave/metal)
  target_link_libraries(codec-layout-tests PRIVATE nonary-framing)
  add_test(NAME codec-layout COMMAND codec-layout-tests)
  add_executable(dequant-fixture-tests tests/dequant_fixture_tests.cpp third_party/pyrowave/metal/pyrowave_bitstream.cpp)
  target_include_directories(dequant-fixture-tests PRIVATE third_party/pyrowave/metal)
  target_link_libraries(dequant-fixture-tests PRIVATE nonary-framing)
  add_test(NAME dequant-fixture COMMAND dequant-fixture-tests)
endif()
option(BUILD_METAL_VIDEO_TOOL "Build native PyroWave encode/decode video harness on macOS" ${APPLE})
if(BUILD_METAL_VIDEO_TOOL)
  if(NOT APPLE)
    message(FATAL_ERROR "Metal video tool requires macOS/Xcode; disable BUILD_METAL_VIDEO_TOOL for portable tests.")
  endif()
  enable_language(OBJCXX)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/prepare_metal.py
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} OUTPUT_VARIABLE PW_METAL_SOURCE
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  add_subdirectory(${PW_METAL_SOURCE} ${CMAKE_BINARY_DIR}/pyrowave-metal EXCLUDE_FROM_ALL)
  add_executable(pyrowave-video tools/pyrowave_video.mm third_party/pyrowave/yuv4mpeg.cpp)
  target_include_directories(pyrowave-video PRIVATE ${PW_METAL_SOURCE} third_party/pyrowave)
  target_compile_options(pyrowave-video PRIVATE -fobjc-arc -Wall -Wextra)
  target_link_libraries(pyrowave-video PRIVATE pyrowave-metal nonary-framing "-framework Foundation" "-framework Metal")
endif()
